Production Window Function CTE for Cash Rollforward

PostgreSQL / Snowflake / T-SQL Compatible 
WITH MonthlyAggregates AS (
    SELECT 
        DATE_TRUNC('month', TransactionDate) AS PeriodDate,
        Channel,
        SUM(CASE WHEN TransactionType = 'Credit' THEN TransactionAmount ELSE 0 END) AS MonthlyCredits,
        SUM(CASE WHEN TransactionType = 'Debit' THEN TransactionAmount ELSE 0 END) AS MonthlyDebits
    FROM Fact_Transactions
    GROUP BY 1, 2
)
SELECT 
    PeriodDate,
    Channel,
    MonthlyCredits,
    MonthlyDebits,
    (MonthlyCredits - MonthlyDebits) AS NetCashFlow,
    SUM(MonthlyCredits - MonthlyDebits) OVER (PARTITION BY Channel ORDER BY PeriodDate) AS RunningBalance
FROM MonthlyAggregates;

Analytical Staging Layer 
WITH ProductMetrics AS (
    SELECT 
        ProductID,
        SUM(CASE WHEN Period = 'M1' THEN UnitsSold ELSE 0 END) AS Q1,
        SUM(CASE WHEN Period = 'M2' THEN UnitsSold ELSE 0 END) AS Q2,
        AVG(CASE WHEN Period = 'M1' THEN NetPrice ELSE NULL END) AS P1,
        AVG(CASE WHEN Period = 'M2' THEN NetPrice ELSE NULL END) AS P2,
        AVG(CASE WHEN Period = 'M1' THEN UnitCOGS ELSE NULL END) AS C1,
        AVG(CASE WHEN Period = 'M2' THEN UnitCOGS ELSE NULL END) AS C2
    FROM Fact_SalesInvoices
    GROUP BY ProductID
)
SELECT 
    ProductID,
    SUM((P2 - P1) * Q2) AS PriceVariance,
    SUM((Q2 - Q1) * (P1 - C1)) AS VolumeVariance,
    SUM((C2 - C1) * Q2) AS CostInflationVariance
FROM ProductMetrics
GROUP BY ProductID;
